cmake_minimum_required(VERSION 3.16)
project(var_chess_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(chessembly chessembly.cpp)
target_include_directories(chessembly PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(chess_engine main.cpp)
target_link_libraries(chess_engine PRIVATE chessembly)
add_executable(chess_tests tests.cpp)
target_link_libraries(chess_tests PRIVATE chessembly)
add_executable(interpreter_tests interpreter_tests.cpp)
target_link_libraries(interpreter_tests PRIVATE chessembly)
enable_testing()
add_test(NAME standard_chess COMMAND chess_tests)
add_test(NAME chessembly COMMAND interpreter_tests)
